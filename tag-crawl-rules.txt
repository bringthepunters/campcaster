# IMPLEMENTATION PSEUDOCODE FOR CAMPGROUND CRAWLER

class CampgroundPageCrawler:
    def extract_campground_data(self, html_content, url):
        """
        Main extraction function for a single campground page
        """
        data = {
            'campground_id': self._extract_id(url),
            'campground_name': self._extract_title(html_content),
            'primary_landscape': None,
            'water_context': None,
            'features': [],
            'fauna_observations': [],  # START EMPTY - only fill if found
            'data_sources': {}
        }
        
        # 1. Extract landscape from description
        description = self._find_description_section(html_content)
        data['primary_landscape'] = self._classify_landscape(description)
        data['data_sources']['landscape_source'] = description[:100] + "..."
        
        # 2. Look for water context
        water_info = self._find_water_references(html_content)
        if water_info:
            data['water_context'] = self._classify_water_context(water_info)
            
        # 3. Extract terrain features from warnings/access
        features_section = self._find_features_warnings(html_content)
        data['features'] = self._extract_terrain_features(features_section)
        
        # 4. CRITICAL: Only extract fauna if wildlife section exists
        wildlife_section = self._find_wildlife_section(html_content)
        if wildlife_section:
            # Found a "Wildlife viewing" or similar section
            fauna_mentions = self._extract_fauna_from_section(wildlife_section)
            data['fauna_observations'] = self._format_fauna_tags(fauna_mentions)
            data['data_sources']['fauna_source'] = "wildlife_section"
        else:
            # NO WILDLIFE SECTION FOUND - leave empty array
            data['fauna_observations'] = []
            data['data_sources']['fauna_source'] = "none_found"
            
        return data
    
    def _find_wildlife_section(self, html):
        """
        Specifically looks for wildlife viewing sections
        """
        # Common section headings in Victorian park pages
        wildlife_headings = [
            "Wildlife viewing",
            "Nature and wildlife", 
            "Birdwatching",
            "Local animals",
            "Wildlife in the area",
            "Animals you might see"
        ]
        
        for heading in wildlife_headings:
            section = self._find_section_by_heading(html, heading)
            if section:
                return section
        return None  # No wildlife section found
    
    def _extract_fauna_from_section(self, section_text):
        """
        Extract specific animal mentions from wildlife section
        """
        # Australian animal name recognition
        australian_fauna_terms = {
            'kangaroo': 'eastern_grey_kangaroo',
            'wallaby': 'swamp_wallaby',
            'wombat': 'common_wombat',
            'koala': 'koala',
            'echidna': 'short-beaked_echidna',
            'kookaburra': 'laughing_kookaburra',
            'rosella': 'crimson_rosella',
            # ... expanded list
        }
        
        found_animals = []
        for common_name, normalized in australian_fauna_terms.items():
            if common_name in section_text.lower():
                found_animals.append(normalized)
                
        return found_animals  # Returns list like ['eastern_grey_kangaroo', 'laughing_kookaburra']
